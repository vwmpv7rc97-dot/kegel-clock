Python 模型转换脚本 (STEP/STL/OBJ → GLTF)

# 安装: pip install trimesh pygltf
import trimesh, os, json

def convert_to_gltf(input_dir, output_dir):
    os.makedirs(output_dir, exist_ok=True)
    for f in os.listdir(input_dir):
        if f.endswith(('.step','.stp','.stl','.obj','.ply')):
            mesh = trimesh.load(os.path.join(input_dir, f))
            mesh.apply_scale(0.001)  # mm → m
            if hasattr(mesh, 'visual'):
                mesh.visual.material = trimesh.visual.material.PBRMaterial(
                    baseColorFactor=[0.7,0.7,0.8,1], metallicFactor=0.8, roughnessFactor=0.4)
            out = os.path.join(output_dir, f.rsplit('.',1)[0]+'.gltf')
            mesh.export(out, file_type='gltf')
            print(f'✅ {f} → {out}')

convert_to_gltf('./cad_models/', './models/')
